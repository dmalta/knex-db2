    - columnInfo() with array-style rows: output([['ID','INTEGER',4,0,'N',null]]) → same result shape
    - columnInfo() with resp.rows format: output({ rows: [{NAME:'ID',...}] }) → same result shape
    - columnInfo() with NULLS:'Y' → nullable:true
    - columnInfo() with DEFAULT:'0' → defaultValue:'0'
    - columnInfo('ID') targeting specific column: output([{NAME:'ID', ...}]) → returns that column's descriptor directly (not the full map)
    - columnInfo() with schema: compiles sql containing "AND TBCREATOR = ?"
    - columnInfo() row with numeric-string key ('0','1',...): output([{'0':'ID','1':'CHAR','2':10,'4':'Y'}]) → parses correctly
  </behavior>
  <action>
    Append a new `describe('QueryCompiler — columnInfo output parsing', ...)` block to
    `tests/unit/schema-compilers.test.js`.

    Get the compiled columnInfo query object via `db.table('users').columnInfo().toSQL()`.
    That returns an object with `{ sql, bindings, output }` where `output` is a function.
    Call `result.output(rows)` directly with different row shapes:

    ```js
    describe('QueryCompiler — columnInfo output parsing', () => {
      function getColumnInfoOutput(columnName) {
        return db.table('users').columnInfo(columnName).toSQL().output;
      }

      const namedRow = { NAME: 'ID', COLTYPE: 'INTEGER', LENGTH: 4, SCALE: 0, NULLS: 'N', DEFAULT: null };
      const arrayRow = ['ID', 'INTEGER', 4, 0, 'N', null];
      const numericKeyRow = { '0': 'ID', '1': 'INTEGER', '2': 4, '3': 0, '4': 'N', '5': null };

      test('named-key rows → correct column descriptor', () => {
        const output = getColumnInfoOutput();
        const cols = output([namedRow]);
        expect(cols.ID).toBeDefined();
        expect(cols.ID.type).toBe('INTEGER');
        expect(cols.ID.nullable).toBe(false);
        expect(cols.ID.maxLength).toBe(4);
      });

      test('array-style rows → correct column descriptor', () => {
        const output = getColumnInfoOutput();
        const cols = output([arrayRow]);
        expect(cols.ID).toBeDefined();
        expect(cols.ID.type).toBe('INTEGER');
      });

      test('numeric string key rows → correct column descriptor', () => {
        const output = getColumnInfoOutput();
        const cols = output([numericKeyRow]);
        expect(cols.ID).toBeDefined();
      });

      test('resp.rows wrapper format → same result', () => {
        const output = getColumnInfoOutput();
        const cols = output({ rows: [namedRow] });
        expect(cols.ID).toBeDefined();
        expect(cols.ID.type).toBe('INTEGER');
      });

      test('NULLS:"Y" → nullable:true', () => {
        const output = getColumnInfoOutput();
        const cols = output([{ ...namedRow, NULLS: 'Y' }]);
        expect(cols.ID.nullable).toBe(true);
      });

      test('DEFAULT value is preserved', () => {
        const output = getColumnInfoOutput();
        const cols = output([{ ...namedRow, DEFAULT: '42' }]);
        expect(cols.ID.defaultValue).toBe('42');
      });

      test('specific column lookup returns that column descriptor directly', () => {
        const output = getColumnInfoOutput('ID');
        const col = output([namedRow]);
        // When a specific column is requested, output() returns the descriptor, not the map
        expect(col.type).toBe('INTEGER');
        expect(col.nullable).toBe(false);
      });

      test('columnInfo with schema includes TBCREATOR binding', () => {
        const origSettings = db.client.connectionSettings;
        db.client.connectionSettings = { currentSchema: 'MYSCHEMA' };
        try {
          const compiled = db.table('users').columnInfo().toSQL();
          expect(compiled.sql).toContain('TBCREATOR');
          expect(compiled.bindings).toContain('MYSCHEMA');
        } finally {
          db.client.connectionSettings = origSettings;
        }
      });
    });
    ```
